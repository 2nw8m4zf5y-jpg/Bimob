cmake_minimum_required(VERSION 3.20)
project(bimob_native_core VERSION 0.2.0 LANGUAGES CXX)
option(BIMOB_NATIVE_BUILD_TESTS "Build native core contract tests" ON)
option(BIMOB_NATIVE_ENABLE_SANITIZERS "Enable ASan and UBSan" OFF)
if(BIMOB_NATIVE_ENABLE_SANITIZERS AND NOT MSVC)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(bimob_foundation src/model.cpp src/engine.cpp src/canonical_json.cpp)
target_include_directories(bimob_foundation PUBLIC include)
target_compile_features(bimob_foundation PUBLIC cxx_std_20)
set_target_properties(bimob_foundation PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(bimob_foundation PRIVATE /W4 /WX)
else()
  target_compile_options(bimob_foundation PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_library(bimob_session_bridge STATIC src/native_session.cpp)
target_link_libraries(bimob_session_bridge PUBLIC bimob_foundation)
target_include_directories(bimob_session_bridge PUBLIC include)
target_compile_features(bimob_session_bridge PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(bimob_session_bridge PRIVATE /W4 /WX)
else()
  target_compile_options(bimob_session_bridge PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(ANDROID)
  add_library(bimob_jni SHARED src/native_bridge.cpp)
  target_link_libraries(bimob_jni PRIVATE bimob_session_bridge)
  target_compile_features(bimob_jni PRIVATE cxx_std_20)
endif()
if(BIMOB_NATIVE_BUILD_TESTS AND NOT ANDROID)
  enable_testing()
  add_executable(bimob_native_contract_tests tests/model_contract_tests.cpp)
  target_link_libraries(bimob_native_contract_tests PRIVATE bimob_foundation)
  target_compile_features(bimob_native_contract_tests PRIVATE cxx_std_20)
  if(MSVC)
    target_compile_options(bimob_native_contract_tests PRIVATE /W4 /WX)
  else()
    target_compile_options(bimob_native_contract_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME native_model_contracts COMMAND bimob_native_contract_tests)
  add_executable(bimob_native_parity_runner tests/native_parity_runner.cpp)
  target_link_libraries(bimob_native_parity_runner PRIVATE bimob_foundation)
  target_compile_features(bimob_native_parity_runner PRIVATE cxx_std_20)
  if(MSVC)
    target_compile_options(bimob_native_parity_runner PRIVATE /W4 /WX)
  else()
    target_compile_options(bimob_native_parity_runner PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME native_parity_all COMMAND bimob_native_parity_runner
    "${CMAKE_SOURCE_DIR}/../shared-tests/golden-scenarios"
    "${CMAKE_BINARY_DIR}/../parity-out/native")
  add_executable(bimob_native_session_tests tests/native_session_tests.cpp)
  target_link_libraries(bimob_native_session_tests PRIVATE bimob_session_bridge)
  target_compile_features(bimob_native_session_tests PRIVATE cxx_std_20)
  if(MSVC)
    target_compile_options(bimob_native_session_tests PRIVATE /W4 /WX)
  else()
    target_compile_options(bimob_native_session_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME native_session_contracts COMMAND bimob_native_session_tests)
endif()
